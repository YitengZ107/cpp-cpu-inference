cmake_minimum_required(VERSION 3.20)
project(cpu_inference VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
option(INFERENCE_NATIVE "Tune for the build CPU (reduces binary portability)" OFF)
option(INFERENCE_SANITIZERS "Enable ASan and UBSan with GCC/Clang" OFF)
add_library(inference src/engine.cpp src/io.cpp)
target_include_directories(inference PUBLIC include)
target_link_libraries(inference PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(inference PRIVATE /W4 /fp:precise)
else()
  target_compile_options(inference PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(INFERENCE_NATIVE)
    target_compile_options(inference PRIVATE -march=native)
  endif()
  if(INFERENCE_SANITIZERS)
    target_compile_options(inference PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(inference PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(inference_cli src/main.cpp)
target_link_libraries(inference_cli PRIVATE inference)
set(INFERENCE_RECORDED_FLAGS "${CMAKE_CXX_FLAGS} $<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:Debug>:${CMAKE_CXX_FLAGS_DEBUG}>$<$<CONFIG:RelWithDebInfo>:${CMAKE_CXX_FLAGS_RELWITHDEBINFO}>$<$<CONFIG:MinSizeRel>:${CMAKE_CXX_FLAGS_MINSIZEREL}>")
if(MSVC)
  string(APPEND INFERENCE_RECORDED_FLAGS " /W4 /fp:precise")
else()
  string(APPEND INFERENCE_RECORDED_FLAGS " -Wall -Wextra -Wpedantic -ffp-contract=off")
  if(INFERENCE_NATIVE)
    string(APPEND INFERENCE_RECORDED_FLAGS " -march=native")
  endif()
  if(INFERENCE_SANITIZERS)
    string(APPEND INFERENCE_RECORDED_FLAGS " -fsanitize=address,undefined -fno-omit-frame-pointer")
  endif()
endif()
string(REPLACE "\\" "\\\\" INFERENCE_RECORDED_FLAGS "${INFERENCE_RECORDED_FLAGS}")
string(REPLACE "\"" "\\\"" INFERENCE_RECORDED_FLAGS "${INFERENCE_RECORDED_FLAGS}")
configure_file(cmake/build_info.hpp.in build_info.hpp.in @ONLY)
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/build_info.hpp
  INPUT ${CMAKE_CURRENT_BINARY_DIR}/build_info.hpp.in)
add_executable(inference_bench src/benchmark.cpp)
target_include_directories(inference_bench PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>)
target_link_libraries(inference_bench PRIVATE inference)
if(BUILD_TESTING)
  add_executable(engine_tests tests/test_engine.cpp)
  target_link_libraries(engine_tests PRIVATE inference)
  add_test(NAME engine_tests COMMAND engine_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_tests COMMAND ${CMAKE_COMMAND} -E env
      "INFERENCE_EXE=$<TARGET_FILE:inference_cli>" ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py)
  endif()
endif()
