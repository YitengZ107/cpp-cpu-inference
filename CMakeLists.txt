cmake_minimum_required(VERSION 3.20)
project(cpu_inference VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
option(INFERENCE_NATIVE "Tune for the build CPU (reduces binary portability)" OFF)
option(INFERENCE_SANITIZERS "Enable ASan and UBSan with GCC/Clang" OFF)
add_library(inference src/engine.cpp)
target_include_directories(inference PUBLIC include)
target_link_libraries(inference PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(inference PRIVATE /W4 /fp:precise)
else()
  target_compile_options(inference PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(INFERENCE_NATIVE)
    target_compile_options(inference PRIVATE -march=native)
  endif()
  if(INFERENCE_SANITIZERS)
    target_compile_options(inference PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(inference PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(engine_tests tests/test_engine.cpp)
  target_link_libraries(engine_tests PRIVATE inference)
  add_test(NAME engine_tests COMMAND engine_tests)
endif()
